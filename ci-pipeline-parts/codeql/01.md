name: Java Security Scan

on:
  pull_request:

permissions:
  contents: read
  security-events: write
  actions: read

jobs:
  codeql:
    name: CodeQL Security Scan
    runs-on: ubuntu-latest

    steps:

      # ------------------------------------------------
      # 1. Checkout source code
      # ------------------------------------------------
      - name: Checkout
        uses: actions/checkout@v4

      # ------------------------------------------------
      # 2. Setup Java
      # ------------------------------------------------
      - name: Setup Java 17
        uses: actions/setup-java@v4
        with:
          java-version: '17'
          distribution: temurin
          cache: maven

      # ------------------------------------------------
      # 3. Initialize CodeQL
      # ------------------------------------------------
      - name: Initialize CodeQL
        uses: github/codeql-action/init@v3
        with:
          languages: java
          queries: security-extended

      # ------------------------------------------------
      # 4. Build Java application
      # ------------------------------------------------
      - name: Build
        run: mvn clean package -DskipTests

      # ------------------------------------------------
      # 5. Analyze with CodeQL
      # ------------------------------------------------
      - name: CodeQL Analysis
        uses: github/codeql-action/analyze@v3
        with:
          category: "/language:java"

      # ------------------------------------------------
      # 6. Enforce security policy
      # ------------------------------------------------
      - name: Enforce CodeQL Security Policy
        env:
          GH_TOKEN: ${{ github.token }}
          REPO: ${{ github.repository }}
        run: |
          echo "Checking CodeQL security findings..."

          alerts=$(gh api \
            -H "Accept: application/vnd.github+json" \
            "/repos/${REPO}/code-scanning/alerts" \
            --paginate)

          high=$(echo "$alerts" | jq '
            [
              .[]
              | select(
                  .state == "open"
                  and
                  (
                    .rule.security_severity_level == "high"
                    or
                    .rule.security_severity_level == "critical"
                  )
              )
            ]
            | length
          ')

          echo "High/Critical CodeQL findings: $high"

          if [ "$high" -gt 0 ]; then
            echo "❌ High/Critical CodeQL vulnerabilities found."
            exit 1
          fi

          echo "✅ No High/Critical CodeQL vulnerabilities found."
