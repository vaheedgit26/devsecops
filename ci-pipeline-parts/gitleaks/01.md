## Git Leaks  
```yaml
- name: Run Gitleaks Scan
  run: |
    docker run --rm \
    -v "${{ github.workspace }}:/repo" \
    zricethezav/gitleaks:latest detect \
    --source=/repo \
    --report-format=json \
    --report-path=/repo/gitleaks-report.json \
    --redact \
    --exit-code 0

- name: Run Gitleaks (Official Image)
  run: |
    docker run --rm \
      -v "${{ github.workspace }}:/repo" \
      ghcr.io/gitleaks/gitleaks:latest \
      git /repo \
      --report-format sarif \
      --report-path /repo/results.sarif \
      --redact \
      --exit-code 0

- name: Upload Gitleaks report
  uses: actions/upload-artifact@v4
  with:
    name: gitleaks-report
    path: gitleaks-report.json
    if-no-files-found: warn
```
**GitHub Actions Flow:**  
```yaml
name: Gitleaks Secret Scan

on:
  pull_request:
    types:
      - opened
      - synchronize
      - reopened

  push:
    branches:
      - main
      - develop

  workflow_dispatch:

permissions:
  contents: read
  pull-requests: read
  security-events: write

concurrency:
  group: gitleaks-${{ github.repository }}-${{ github.ref }}
  cancel-in-progress: true

jobs:
  gitleaks:
    name: Gitleaks Secret Scan
    runs-on: self-hosted

    steps:

      # ==========================================================
      # Checkout complete Git history
      # ==========================================================
      - name: Checkout repository
        uses: actions/checkout@v6
        with:
          fetch-depth: 0

      # ==========================================================
      # Run Gitleaks
      # ==========================================================
      - name: Run Gitleaks
        id: gitleaks
        uses: gitleaks/gitleaks-action@v3
        # continue-on-error: true
        env:
          GITHUB_TOKEN: ${{ secrets.GITHUB_TOKEN }}

          # Uncomment if your organization requires a Gitleaks license
          # GITLEAKS_LICENSE: ${{ secrets.GITLEAKS_LICENSE }}

      # ==========================================================
      # Upload SARIF to GitHub Code Scanning
      # ==========================================================
      - name: Upload Gitleaks SARIF
        if: always()
        uses: github/codeql-action/upload-sarif@v4
        with:
          sarif_file: results.sarif
          category: gitleaks

      # ==========================================================
      # Upload raw SARIF as GitHub Actions artifact
      # ==========================================================
      - name: Upload Gitleaks report artifact
        if: always()
        uses: actions/upload-artifact@v4
        with:
          name: gitleaks-sarif-${{ github.run_number }}
          path: results.sarif
          if-no-files-found: warn
          retention-days: 2
```
> **Note:  Default Gitleaks Github Action code will produce in `results.sarif`**
